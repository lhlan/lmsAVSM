            function isQuizReallyCompleted() {
                // Check 1: Must have results container

                if ($results.length === 0) {
                    return false;

                    return false;

                const hasScoreIndicators = resultsText.includes('%') || 
                                          resultsText.includes('score') || 
                                          resultsText.includes('điểm') ||
                                          resultsText.includes('correct') ||
                                          resultsText.includes('đúng');
                
                if (!hasScoreIndicators) {
                    return false;
                }
                
                // Check 4: Quiz form should be hidden or disabled
                const $quizForm = $('.wpProQuiz_quiz, .ld-quiz-form');
                const isQuizHidden = $quizForm.length === 0 || !$quizForm.is(':visible');
                
                // Check 5: Should have completion message
                const hasCompletionMsg = $('.wpProQuiz_text .wpProQuiz_header, .ld-quiz-complete-text').length > 0;
                
                debug('Completion checks: results=' + $results.length + 
                      ', visible=' + $results.is(':visible') + 
                      ', hasScore=' + hasScoreIndicators + 
                      ', quizHidden=' + isQuizHidden +
                      ', hasMsg=' + hasCompletionMsg);
                
                // Must pass at least 3 checks
                const passedChecks = [
                    $results.length > 0,
                    $results.is(':visible'),
                    hasScoreIndicators,
                    isQuizHidden || hasCompletionMsg
                ].filter(Boolean).length;
                
                return passedChecks >= 3;
            }

                }
            }
            
            // Extract score from quiz result
            function extractQuizScore() {
                let score = 0;
                
                debug('🔎 Attempting to extract quiz score...');
                
                // Log all elements that might contain score
                debug('=== SCANNING FOR SCORE ELEMENTS ===');
                
                // Method 1: Try wpProQuiz result percentage
                const $percentage = $('.wpProQuiz_resultValue, .wpProQuiz_passPercent');
                if ($percentage.length > 0) {
                    $percentage.each(function(i) {
                        debug('wpProQuiz element ' + i + ': ' + $(this).text());
                    });
                    const text = $percentage.first().text();
                    const match = text.match(/(\d+(?:\.\d+)?)\s*%/);
                    if (match) {
                        score = parseFloat(match[1]);
                        debug('✅ Extracted score from wpProQuiz: ' + score, 'success');
                        return score;
                    }
                }
                
                // Method 2: Try LearnDash score elements
                const ldSelectors = [
                    '.ld-quiz-score',
                    '.ld-quiz-result-score',
                    '.ld_quiz_result_score',
                    '.learndash_quiz_result_score',
                    '.ld-propanel-quiz-results-percentage',
                    '.ld_quiz_results_result_percentage'
                ];
                
                for (let selector of ldSelectors) {
                    const $elem = $(selector);
                    if ($elem.length > 0) {
                        $elem.each(function(i) {
                            debug('LD element ' + selector + ' ' + i + ': ' + $(this).text());
                        });
                        const text = $elem.first().text();
                        const match = text.match(/(\d+(?:\.\d+)?)/);
                        if (match) {
                            score = parseFloat(match[1]);
                            debug('✅ Extracted score from ' + selector + ': ' + score, 'success');
                            return score;
                        }
                    }
                }
                
                // Method 3: Search in all elements with class containing "result" or "score"
                debug('Searching in all result/score classes...');
                $('[class*="result"], [class*="score"], [class*="Result"], [class*="Score"]').each(function() {
                    const $this = $(this);
                    const text = $this.text();
                    if (text.length > 0 && text.length < 200) { // Ignore very long texts
                        debug('Found element .' + this.className + ': ' + text.substring(0, 100));
                        
                        // Try to extract percentage
                        const percentMatch = text.match(/(\d+(?:\.\d+)?)\s*%/);
                        if (percentMatch && score === 0) {
                            const possibleScore = parseFloat(percentMatch[1]);
                            if (possibleScore >= 0 && possibleScore <= 100) {
                                score = possibleScore;
                                debug('✅ Found score with %: ' + score, 'success');
                            }
                        }
                    }
                });
                
                if (score > 0) {
                    return score;
                }
                
                // Method 4: Calculate from correct/total answers
                debug('Attempting to calculate from answers...');
                
                // Try different correct answer indicators
                const correctSelectors = [
                    '.wpProQuiz_correct',
                    '.ld-answer-correct',
                    '.learndash-answer-correct',
                    '.wpProQuiz_answerCorrect',
                    'li.wpProQuiz_answerCorrect',
                    '[class*="correct"]'
                ];
                
                let $correctAnswers = $();
                for (let selector of correctSelectors) {
                    const $found = $(selector);
                    if ($found.length > 0) {
                        $correctAnswers = $found;
                        debug('Found ' + $found.length + ' correct answers with selector: ' + selector);
                        break;
                    }
                }
                
                // Try different question list selectors
                const questionSelectors = [
                    '.wpProQuiz_list',
                    '.wpProQuiz_listItem',
                    '.ld-question-list li',
                    '.ld-question-item',
                    '.learndash-question',
                    '.wpProQuiz_question'
                ];
                
                let totalQuestions = 0;
                for (let selector of questionSelectors) {
                    const $found = $(selector);
                    if ($found.length > 0) {
                        totalQuestions = $found.length;
                        debug('Found ' + totalQuestions + ' questions with selector: ' + selector);
                        break;
                    }
                }
                
                if (totalQuestions > 0 && $correctAnswers.length > 0) {
                    score = Math.round(($correctAnswers.length / totalQuestions) * 100);
                    debug('✅ Calculated score: ' + $correctAnswers.length + '/' + totalQuestions + ' = ' + score + '%', 'success');
                    return score;
                }
                
                // Method 5: Try to find statistic box
                debug('Checking statistic elements...');
                const $statistic = $('.wpProQuiz_catOverview, .ld_quiz_results, .ld-quiz-results-statistics');
                if ($statistic.length > 0) {
                    const statText = $statistic.text();
                    debug('Statistic text: ' + statText.substring(0, 200));
                    
                    // Look for patterns like "10 of 10" or "10/10"
                    const fractionMatch = statText.match(/(\d+)\s*(?:of|\/)\s*(\d+)/i);
                    if (fractionMatch) {
                        const correct = parseInt(fractionMatch[1]);
                        const total = parseInt(fractionMatch[2]);
                        score = Math.round((correct / total) * 100);
                        debug('✅ Calculated from fraction: ' + correct + '/' + total + ' = ' + score + '%', 'success');
                        return score;
                    }
                }
                
                // Method 6: Check for quiz result data attributes
                debug('Checking data attributes...');
                const $quizResults = $('.wpProQuiz_results, .ld-quiz-results, [class*="quiz-result"]');
                if ($quizResults.length > 0) {
                    debug('Quiz results HTML: ' + $quizResults.html().substring(0, 500));
                    
                    // Try data attributes
                    const dataScore = $quizResults.data('score') || $quizResults.data('percentage') || $quizResults.data('result');
                    if (dataScore) {
                        score = parseFloat(dataScore);
                        debug('✅ Found score in data attribute: ' + score, 'success');
                        return score;
                    }
                }
                
                // Method 7: Look in the entire quiz container
                debug('Searching entire quiz container...');
                const $quizContainer = $('.wpProQuiz_content, .learndash-wrapper, .ld-quiz-wrapper');
                if ($quizContainer.length > 0) {
                    const allText = $quizContainer.text();
                    debug('Quiz container text sample: ' + allText.substring(0, 300));
                    
                    // Try to find any percentage
                    const allMatches = allText.match(/(\d+(?:\.\d+)?)\s*%/g);
                    if (allMatches && allMatches.length > 0) {
                        debug('Found percentages: ' + allMatches.join(', '));
                        // Take the first valid percentage
                        for (let match of allMatches) {
                            const num = parseFloat(match);
                            if (num >= 0 && num <= 100) {
                                score = num;
                                debug('✅ Using percentage: ' + score + '%', 'success');
                                break;
                            }
                        }
                    }
                }
                
                if (score === 0) {
                    debug('⚠️ Could not extract score, returning 0', 'error');
                    debug('PLEASE HELP: Open console and run this command:');
                    debug('jQuery(".wpProQuiz_results, .ld-quiz-results").html()');
                }
                
                return score;
            }
            
        // Normalize different event payload shapes into a percentage [0, 100]
        function normalizeScoreFromEvent(data) {
            if (!data || typeof data !== 'object') {
                return 0;
            }

            const toNumber = (v) => {
                const n = Number(v);
                return Number.isFinite(n) ? n : NaN;
            };

            // 1) Prefer explicit percentage-like fields
            const percentCandidates = [
                data.percentage,
                data.pass_percent,
                data.passPercent,
                data.percent,
                data.prozent
            ].map(toNumber).filter((n) => Number.isFinite(n));

            for (const p of percentCandidates) {
                // Handle 0-1 fractions as well as 0-100 values
                if (p > 0 && p <= 1) return Math.round(p * 100);
                if (p >= 0 && p <= 100) return Math.round(p);
            }

            // 2) Try points-based fields with a max
            const rawScore = toNumber(
                data.score ?? data.points ?? data.user_points ?? data.earned_points
            );
            const maxPoints = toNumber(
                data.max_score ?? data.maxPoints ?? data.total_points ?? data.points_total ?? data.pointsPossible ?? data.totalPoints ?? data.score_max
            );

            if (Number.isFinite(rawScore)) {
                // If we know the maximum, compute percentage reliably
                if (Number.isFinite(maxPoints) && maxPoints > 0) {
                    return Math.round((rawScore / maxPoints) * 100);
                }

                // Handle normalized fractions
                if (rawScore > 0 && rawScore <= 1) {
                    return Math.round(rawScore * 100);
                }

                // If the score looks too large, attempt to infer from question counts
                const totalQuestions = toNumber(
                    data.total_questions ?? data.questionsCount ?? data.questionCount ?? data.totalQuestions
                );
                if (rawScore > 100 && Number.isFinite(totalQuestions) && totalQuestions > 0) {
                    // Heuristic: many payloads use points-per-question; infer points per question
                    const pointsPerQuestion = Math.round(rawScore / totalQuestions);
                    if (pointsPerQuestion > 0) {
                        return Math.round((rawScore / (totalQuestions * pointsPerQuestion)) * 100);
                    }
                }

                // Fall back: if it's in [0,100], use it; otherwise NaN and let caller fallback
                if (rawScore >= 0 && rawScore <= 100) {
                    return Math.round(rawScore);
                }
            }

            return 0;
        }
        
            // Handle quiz completion
            function handleQuizCompletion(data) {
                // Prevent duplicate handling
                if (quizCompletionHandled) {
                    debug('⚠️ Quiz completion already handled, skipping', 'info');
                    return;
                }
                debug('📊 Processing quiz completion...', 'info');
                
            // Try to read and normalize from event payload first
            let score = normalizeScoreFromEvent(data);
            debug('Score from event data (normalized): ' + score);

            // If event payload didn't give a valid score, fallback to DOM extraction
            if (!Number.isFinite(score) || score <= 0 || score > 100) {
                debug('Event score invalid/out-of-range, will re-check DOM shortly...');
                // Give the UI time to render results, then perform a manual check
                setTimeout(function() {
                    if (!quizCompletionHandled) {
                        checkForQuizCompletion();
                    }
                }, 1500);
                return; // Defer handling until DOM has results
            }

            // Final clamp to [0, 100]
            score = Math.max(0, Math.min(100, Math.round(Number(score) || 0)));

            // Mark as handled now that we have a valid score
            quizCompletionHandled = true;

            debug('✅ Final score: ' + score + '%', 'success');
                
                // Show loading message
                alert('⏳ Đang lưu kết quả...', 'info');
                
                // Update score on server
                const updateData = {
                    action: 'ld_save_student_v3',
                    fullname: studentData.fullname,
                    email: studentData.email,
                    phone: studentData.phone,
                    quiz_id: studentData.quiz_id,
                    score: score
                };
                
                debug('Sending score update to server...');
                debug('Update data: ' + JSON.stringify(updateData));
                
                $.ajax({
                    url: ajaxUrl,
                    type: 'POST',
                    data: updateData,
                    dataType: 'json',
                    timeout: 15000,
                    success: function(response) {
                        debug('✅ Score updated successfully!', 'success');
                        debug('Server response: ' + JSON.stringify(response));
                        
                        alert('✅ Đã lưu kết quả thành công!', 'success');
                        
                        // Wait a bit before hiding quiz
                        setTimeout(function() {
                            // Hide quiz
                            $quizContent.slideUp(500);
                            
                            // Show result
                            displayResult(studentData, score);
                        }, 1000);
                    },
                    error: function(xhr, status, error) {
                        debug('❌ Failed to update score', 'error');
                        debug('Error: ' + status + ' - ' + error, 'error');
                        
                        alert('⚠️ Lỗi khi lưu điểm. Vui lòng chụp màn hình kết quả.', 'error');
                        
                        // Still show result even if save failed
                        displayResult(studentData, score);
                    }
                });
            }
            
            // Display result
            function displayResult(student, score) {
                const $result = $wrapper.find('.ld-result');

            }
            
            // Escape HTML
            function escapeHtml(text) {
                const map = {
                    '&': '&amp;',
                    '<': '&lt;',
                    '>': '&gt;',
                    '"': '&quot;',
                    "'": '&#039;'
                };
                return String(text).replace(/[&<>"']/g, function(m) { return map[m]; });
            }

add_action('wp_ajax_ld_save_student_v3', 'ld_save_student_v3_handler');
add_action('wp_ajax_nopriv_ld_save_student_v3', 'ld_save_student_v3_handler');

function ld_save_student_v3_handler() {
    error_log('=== LD QUIZ V3 SAVE START ===');
    error_log('POST data: ' . print_r($_POST, true));
    
    try {
        // Get data
        $fullname = isset($_POST['fullname']) ? sanitize_text_field($_POST['fullname']) : '';
        $email = isset($_POST['email']) ? sanitize_email($_POST['email']) : '';
        $phone = isset($_POST['phone']) ? sanitize_text_field($_POST['phone']) : '';
        $quiz_id = isset($_POST['quiz_id']) ? intval($_POST['quiz_id']) : 0;
        $score = isset($_POST['score']) ? sanitize_text_field($_POST['score']) : '0';
        
        error_log("Parsed: name=$fullname, email=$email, phone=$phone, quiz=$quiz_id, score=$score");
        
        // Validate
        if (empty($fullname) || empty($email) || empty($phone)) {
            error_log('Validation failed: missing fields');
            wp_send_json_error(array(
                'message' => 'Vui lòng điền đầy đủ thông tin'
            ));
        }
        
        // Google Sheets URL
        $sheets_url = "https://script.google.com/macros/s/AKfycbyWZ_WWGRieWywpTweI5alSYRQGskk_OmIfGGAbEU3lsU75F5Z88Ro4hNDH8wvAjFki1A/exec";
        
        error_log('Sending to Google Sheets: ' . $sheets_url);
        
        $post_data = array(
            'name' => $fullname,
            'email' => $email,
            'mobiphone' => $phone,
            'quiz_id' => $quiz_id,
            'score' => $score,
            'timestamp' => current_time('mysql'),
            'ip' => $_SERVER['REMOTE_ADDR']
        );
        
        error_log('Payload: ' . print_r($post_data, true));
        
        $response = wp_remote_post($sheets_url, array(
            'timeout' => 20,
            'body' => $post_data
        ));
        
        if (is_wp_error($response)) {
            $error = $response->get_error_message();
            error_log('Google Sheets error: ' . $error);
            wp_send_json_error(array(
                'message' => 'Lỗi kết nối Google Sheets: ' . $error
            ));
        }
        
        $http_code = wp_remote_retrieve_response_code($response);
        $body = wp_remote_retrieve_body($response);
        
        error_log('Google Sheets HTTP code: ' . $http_code);
        error_log('Google Sheets response: ' . substr($body, 0, 500));
        
        if ($http_code !== 200) {
            error_log('Google Sheets HTTP error');
            wp_send_json_error(array(
                'message' => 'Google Sheets HTTP error: ' . $http_code
            ));
        }
        
        error_log('=== LD QUIZ V3 SAVE SUCCESS ===');
        
        wp_send_json_success(array(
            'message' => 'Saved successfully',
            'student' => array(
                'name' => $fullname,
                'email' => $email,
                'phone' => $phone
            )
        ));
        
    } catch (Exception $e) {
        error_log('Exception: ' . $e->getMessage());
        error_log($e->getTraceAsString());
        wp_send_json_error(array(
            'message' => 'Exception: ' . $e->getMessage()
        ));
    }
}
