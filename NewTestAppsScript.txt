function doPost(e) {
  try {
    // Log request để debug
    Logger.log('=== REQUEST RECEIVED ===');
    Logger.log('Content Type: ' + e.postData.type);
    Logger.log('Raw content: ' + e.postData.contents);
    
    // Lấy parameters
    var params = e.parameter;
    Logger.log('Parameters: ' + JSON.stringify(params));
    
    // Lấy dữ liệu từ request
    var name = params.name || '';
    var email = params.email || '';
    var mobiphone = params.mobiphone || '';
    var score = params.score || '0';
    var quiz_id = params.quiz_id || '';
    var timestamp = params.timestamp || '';
    var ip = params.ip || '';
    
    // Validate dữ liệu
    if (!name || !email || !mobiphone) {
      Logger.log('Validation failed: Missing required fields');
      return createResponse(false, 'Missing required fields: name, email, or phone');
    }
    
    // Lấy spreadsheet hiện tại
    //var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var sheet = SpreadsheetApp.openById("1Gl3-SCrbZEFkaMQimFuIuQO-ssmw9BbS8tjHzOHBGgE").getSheetByName("Sheet1");
    // Nếu sheet trống, thêm header
    if (sheet.getLastRow() === 0) {
      sheet.appendRow([
        'Thời gian',
        'Họ tên',
        'Email', 
        'Số điện thoại',
        'Điểm số',
        'Quiz ID',
        'IP Address',
        'Timestamp từ WP'
      ]);
      
      // Format header
      var headerRange = sheet.getRange(1, 1, 1, 8);
      headerRange.setFontWeight('bold');
      headerRange.setBackground('#4285f4');
      headerRange.setFontColor('#ffffff');
    }
    
    // Thêm dữ liệu mới
    var currentTime = new Date();
    sheet.appendRow([
      currentTime,           // Thời gian server
      name,                  // Họ tên
      email,                 // Email
      mobiphone,             // Số điện thoại
      score,                 // Điểm số
      quiz_id,               // Quiz ID
      ip,                    // IP Address
      timestamp              // Timestamp từ WordPress
    ]);
    
    // Format row vừa thêm
    var lastRow = sheet.getLastRow();
    var dataRange = sheet.getRange(lastRow, 1, 1, 8);
    
    // Alternate row colors
    if (lastRow % 2 === 0) {
      dataRange.setBackground('#f8f9fa');
    }
    
    // Auto-resize columns
    for (var i = 1; i <= 8; i++) {
      sheet.autoResizeColumn(i);
    }
    
    Logger.log('=== DATA SAVED SUCCESSFULLY ===');
    Logger.log('Row: ' + lastRow);
    
    // Return success response
    return createResponse(true, 'Data saved successfully', {
      row: lastRow,
      name: name,
      email: email
    });
    
  } catch (error) {
    Logger.log('=== ERROR ===');
    Logger.log('Error message: ' + error.message);
    Logger.log('Error stack: ' + error.stack);
    
    return createResponse(false, 'Server error: ' + error.message);
  }
}
